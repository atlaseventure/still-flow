add_executable(stillflow main.cpp)
target_link_libraries(stillflow PRIVATE StillFlow::desktop_c)
target_compile_options(stillflow PRIVATE $<$<COMPILE_LANG_AND_ID:CXX,GNU,Clang>:-Wall;-Wextra;-Wpedantic>)
if(NOT CMAKE_RUNTIME_OUTPUT_DIRECTORY)
    set_target_properties(stillflow PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}")
endif()
set_target_properties(stillflow PROPERTIES INSTALL_RPATH "$ORIGIN/../${CMAKE_INSTALL_LIBDIR}")
install(TARGETS stillflow RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
